## My solution


function balanced(input) {
    var stack = [];

    var matching = {
        "}": "{",
        ")": "(",
        "]": "["
    };

    for (var pos = 0; pos < input.length; pos++) {
        var char = input.charAt(pos);

        if ("{([".includes(char)) {
            stack.push(char);
        }

        else if ("})]".includes(char)) {
            var lastOpening = stack.pop();

            if (lastOpening != matching[char]) {
                return "Mismatch at position " + pos;
            }
        }
    }

    if (stack.length > 0) {
        return "Missing closing bracket";
    }

    return "Balanced";
}
